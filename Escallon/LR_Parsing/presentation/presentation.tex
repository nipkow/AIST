\section{Basic Definitions}

\subsection{Extended and Reduced Grammars}

\begin{frame}
  \frametitle{Reduced and Extended Grammars}
  \begin{definition}[Reduced grammar]
    A CFG $G$ is \concept{reduced} if its productions contain only useful
    nonterminals.
  \end{definition}
  \begin{definition}[Extended grammar]
    Let $G$ be a CFG with production set $P$ and start symbol $S$.
    The \concept{extension} of $G$ is the CFG resulting from extending $G$ by a
    new start symbol $S'$, and $P$ by the production $S' \rightarrow S$. This resulting
    grammar is therefore called \concept{extended}.
  \end{definition}
  \begin{block}{Convention}
    \begin{itemize}
      \item $G$ is a CFG with start symbol $S$ production set $P$, terminals $V_T$
      and nonterminals $V_N$
      \item $G'$ is the extension of $G$ with start symbol $S'$, production set $P'$,
      and nonterminals $V_N'$
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Context-Free Items}

\begin{frame}
  \frametitle{Context-Free Items}
  \begin{definition}
    A context-free item for $G$ is a triple 
    $(A, \alpha, \beta) \in V_N \times \syms \times \syms$ such that
    $(A, \alpha @ \beta) \in P$.\\
    The item $(A, \alpha, \beta)$ is often written as \cfitem{A}{\alpha}{\beta}
  \end{definition}
  \begin{itemize}
    \item Context-free items allow tracking the current state of the parsing process
    \item Symbols to the right of the bullet are shifted to the left as they are consumed
    \item Empty strings are often written implicitly
    \begin{itemize}
      \item Example: \cfitem{A}{}{\beta} instead of \cfitem{A}{\varepsilon}{\beta}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Further Definitions}
  \begin{itemize}
    \item An item of the form \cfitem{A}{\alpha}{} is \concept{complete}
    \item The \concept{history} of an item of the form \cfitem{A}{\alpha}{\beta} is $\alpha$
    \item If $I$ is a set of items, $completes\ I$ denotes the set of complete items
    in $I$
    \item If $\varrho$ is a list of items, $hist\ \varrho$ is the concatenation of the
    histories of the items in $\varrho$
    \item \It\ denotes the set of all items corresponding to the CFG G
  \end{itemize}
\end{frame}

\subsection{Generalized Pushdown Automata}

\begin{frame}
  \frametitle{Generalized Pushdown Automata}
  \begin{itemize}
    \item Multiple automata that use a stack to operate will be defined
    \item Reading multiple stack symbols in a single transition sometimes necessary
    \item \concept{Generalized pushdown automata} behave like conventional PDAs but
    are able to read topmost stack strings
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{GPDA Configurations}
  \begin{itemize}
    \item A \concept{configuration} for a GPDA with states $Q$ and alphabet $\varSigma$ 
    is a tuple
    $(qs, w) \in Q^+ \times \varSigma^*$
    \begin{itemize}
      \item $qs$: current stack 
      \item $w$: remaining input
    \end{itemize}
    \item A configuration for a GPDA $M$ with initial state $q_0$ is
    \concept{initial} if the stack 
    consists of the list $[q_0]$
    \item If the set of final states of $M$ is F, a \concept{final configuration}
    is of the form $([f], \varepsilon)$ for $f \in F$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{GPDA Steps}
  \begin{itemize}
    \item \concept{computations} = sequences of GPDA steps
    \begin{itemize}
      \item denoted by the reflexive transitive closure of steps, written as $\vdash^*$
    \end{itemize}
    \item computations of length $n$ denoted by $\vdash^n$
    \item The language of a GPDA $M$ with set of final states $F$ and initial state
    $q_0$:
    \[L(M) = \{w \mid \exists f \in F.\ ([q_0], w) \vdash^* ([f], \varepsilon)\}\]
  \end{itemize}
\end{frame}

    and nonterminals $V_N$
    \item $G'$ is the extension of $G$ with start symbol $S'$, production set $P'$,

    \item $P$ is finite
    \item $L(G)$ is nonempty
    \item $G$ is reduced
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{The Item Pushdown Automaton}
  \begin{definition}
    The \concept{item pushdown automaton} (IPDA) to G is the GPDA \ipda\ with
    alphabet $V_T$ and
    \begin{align*}
      states &= \It[G']\\
      init &= \cfitem{S'}{}{S}\\
      final &= \{\cfitem{S'}{S}{}\}\\
      nxt &=
    \begin{multlined}[t]
        \{([\cfitem{X}{\beta}{a \gamma}], a, [\cfitem{X}{\beta a}{\gamma}])\\
        \mid X\ \beta\ a\ \gamma.\ (X, \beta a \gamma) \in P'\}
    \end{multlined}\\
      eps &= \ldots
    \end{align*}
  \end{definition}
  \begin{itemize}
    \item IPDA steps notated by $\vdash_\var{I}$, $\vdash^{*}_{\var{I}}$, and $\vdash^{n}_{\var{I}}$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{IPDA Transitions}
  \begin{itemize}
    \item \concept{shifting} transitions ($nxt$)
    \[ (\cfitem{A}{\alpha}{a \beta} \varrho, aw) \vdash_{\var{I}} (\cfitem{A}{\alpha a}{\beta}\varrho, w) \]
    \item \concept{expanding} transitions
    \[ (\cfitem{A}{\alpha}{B \beta} \varrho, w) \vdash_{\var{I}} (\cfitem{b}{}{\gamma}
    \cfitem{A}{\alpha}{B \beta} \varrho, w) \] 
    \item \concept{reducing} transitions
    \[(\cfitem{B}{\gamma}{} \cfitem{A}{\alpha}{B \beta} \varrho, w) \vdash_{\var{I}}
    (\cfitem{A}{\alpha B}{\beta} \varrho, w)\] 
  \end{itemize}
\end{frame}

\section{Rightmost Chains}

\begin{frame}
  \frametitle{Rightmost Chains}
  \framesubtitle{Motivation}
  \begin{quote}
    Let us now assume that we have a rightmost derivation 
    $S' \deriversa \gamma' A w \derivera \gamma' \alpha \beta$. This derivation
    can be decomposed into
    \begin{align*}
    S' &\derivera \alpha_{1} X_{1} \beta_{1} \deriversa \alpha_{1} X_{1} v_{1}\\
    &\deriversa \ldots \deriversa (\alpha_{1} \ldots \alpha_{n}) X_n (v_n \ldots v_1)\\
    &\derivera (\alpha_{1} \ldots \alpha_{n}) \alpha \beta (v_n \ldots v_1)
    \end{align*}
    for $X_n = A$.
  \end{quote}
  \begin{flushright}
  --- Wilhelm et al., p.~107
  \end{flushright}
\end{frame}

\begin{frame}
  \frametitle{Rightmost Chains}
  \framesubtitle{Definition}
  \begin{itemize}
    \item \chain{P}{\alpha}{\varrho}{\beta}: $\alpha$ reaches $\beta$ with
    rightmost chain $\varrho$ under production set $P$
  \end{itemize}
  \pause
  \begin{gather*}
    \inferrule[refl]{}{\chain{P}{\alpha}{\varepsilon}{\alpha}}\\[10pt]
    \inferrule[step]{
      \chain{P}{\alpha_0}{\varrho}{\alpha X v} \\
      \deriver{P}{\alpha X v}
        {\alpha \alpha' Y \beta v} \\
      \derivers{P}{\beta}{u}
      }{\chain{P}{\alpha_0}{\cfitem{X}{\alpha'}{Y \beta} \varrho}{\alpha \alpha' Y u v}}
  \end{gather*}
\end{frame}

\begin{frame}
  \frametitle{Properties of Rightmost Chains}
  \begin{lemma}[Rightmost chains imply rightmost derivations]
    If \chain{P}{\alpha}{\varrho}{\beta}, then \derivers{P}{\alpha}{\beta}.
  \end{lemma}
  \begin{lemma}[Rightmost derivations imply rightmost chains]\label{deriver_rm}
    If \derivern{P}{A}{n+1}{\alpha X v}, then there
    exists a rightmost chain of the form
    \[ \chain{P}{A}{\cfitem{B}{\alpha'}{X \beta} \varrho}{\alpha X v}. \]
  \end{lemma}
  \begin{lemma}[Correspondence of IPDA stacks to rightmost chains]\label{ipda_rm}
    If $(\cfitem{A}{\alpha}{\beta} \varrho, w)
        \vdash_\var{I}^* ([\cfitem{S'}{S}{}], \varepsilon)$, then either
        $\varrho = \varepsilon$ or
    there exist $\sigma, X, \alpha', \beta'$, and $\gamma$ such that 
     $\varrho = \cfitem{X}{\alpha'}{A \beta'} \sigma
      \text{ and } \chain{P'}{S'}{\varrho}{\gamma}.$
  \end{lemma}
\end{frame}

\begin{frame}\label{thm341}
  \frametitle{Theorem 3.4.1: the Relation between \charfa, Rightmost Derivations, 
  and \ipda}
  \note{ 
  \begin{itemize}
    \item (1) → (2)
    \begin{itemize}
      \item char(G) can read S from input without performing epsilon transitions
      \item IH applied for a general variable when it holds only for the
      empty list
    \end{itemize}
    \item $w$ is the same in direction (2) → (3) (plus: assm beta derives v) ,
    not otherwise
  \end{itemize}
  }
  \begin{quote}
    Let $G$ be a CFG and $\gamma \in \symse $. The following three
    statements are equivalent:
    \begin{enumerate}
      \item There exists a computation
      $(\cfitem{S'}{}{S}, \gamma) \vdash_c^* (\cfitem{A}{\alpha}{\beta}, \varepsilon)$
      of the characteristic finite automaton \charfa.
      \item There exists a rightmost derivation 
      \derivers{P'}{S'}{\gamma' A w}
      $\derivera \gamma' \alpha \beta w$ with
      $\gamma = \gamma' \alpha$
      \item There exists a computation 
      $(\cfitem{A}{\alpha}{\beta} \varrho, w)
        \vdash_\var{I}^* ([\cfitem{S'}{S}{}] ,\varepsilon)$ of the IPDA \ipda\ such 
        that $\gamma = hist\ (rev\ \varrho) \alpha$ holds.
    \end{enumerate}
  \end{quote}
  \begin{itemize}
    \item Errors corrected
    \item Theorem of equivalence weaker, but individual implications unchanged
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Correcting Theorem 3.4.1}
  \note{Remark use of rightmost chains for bottom two lemmas}
  \small
  \begin{lemma}[$(1) \implies (2)$]
    If there exists a computation
      $(\cfitem{S'}{}{S}, \gamma) \vdash_c^* (\cfitem{A}{\alpha}{\beta}, \varepsilon)$
      of \charfa, then there exists a rightmost derivation of the form
      \derivers{P'}{S'}{\gamma' A w}
      $\derivera \gamma' \alpha \beta w$ with
      $\gamma = \gamma' \alpha$.
  \end{lemma}
  \begin{lemma}[$(2) \implies (3)$]
    If there exists a rightmost derivation
      \derivers{P'}{S'}{\gamma' A w}
      $\derivera \gamma' \alpha \beta w$
      with $\derives{P'}{\beta}{v}$, there exists an IPDA computation
      $(\cfitem{A}{\alpha}{\beta} \varrho, v w)
        \vdash_\var{I}^* ([\cfitem{S'}{S}{}] ,\varepsilon)$ with
        $\gamma' = hist\ (rev\ \varrho)$.
  \end{lemma}
  \begin{lemma}[$(3) \implies (1)$]
    If there exists an IPDA computation 
      $(\cfitem{A}{\alpha}{\beta} \cons \varrho, w)
        \vdash_\var{I}^* ([\cfitem{S'}{S}{}] ,\varepsilon)$ 
    then there exists a \charfa\ computation of the form
      $(\cfitem{S'}{}{S}, hist\ (rev \varrho) \alpha)
        \vdash_c^* (\cfitem{A}{\alpha}{\beta}, \varepsilon)$.
  \end{lemma}
\end{frame}

\section{The $LR(0)$ Parser}

\subsection{Definition}

\begin{frame}
  \frametitle{The $LR(0)$ Parser}
  \note{
    \begin{itemize}
      \item States restricted to Q explicitly akin to the IPDA
      \item Extra: X neq S' avoids conflicts between finish and reduce
      \item f is arbitrary item not in Q
    \end{itemize}
  }
  \begin{definition}
    Let $\varDelta_G$ be the transition function of \lrdfa, $q_{0,G}$ the initial
    state of \lrdfa, $Q$ the set of states of \lrdfa, and $f$ the set
    $\{\cfitem{S'}{}{}\}$. The \concept{$LR(0)$ parser} to G is the 
    GPDA \parser\ with alphabet $V_T$ and 
    \vspace{-5pt}
    \begin{align*}
      states &= Q \cup \{f\}\\
      init &= q_{0,G}\\
      final &= \{f\}\\
      nxt &= \{([q], a, [\varDelta_G(q,a), q])
        \mid q \in Q \land \varDelta_G(q,a) \neq \emptyset\}\\
      \text{and } eps &= \ldots
    \end{align*}
  \end{definition}
  \begin{itemize}
    \item Parser steps denoted by $\vdash_P$, $\vdash_P^*$, and $\vdash_P^n$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{\parser\ Transitions}
  \begin{itemize}
    \item \concept{Reading} transitions ($nxt$)
    \[ (\cfitem{A}{\alpha}{a \beta} \varrho, aw) \vdash_P (\cfitem{A}{\alpha a}{\beta}, w) \]
    \item \concept{Reducing} transitions
    \begin{gather*}
    (q_n \ldots q_1 q \varrho, w) \vdash_P (\varDelta_G(q, X) q \varrho, w)\\
    \text{ if } \cfitem{X}{\alpha}{} \in q_n \text{ with } |\alpha| = n
    \end{gather*}
    \item {Finishing} transitions
    \begin{gather*}
      (q q_0 \varrho, w) \vdash_P (f \varrho, w)\\
      \text{ if } \cfitem{S'}{S}{} \in q
    \end{gather*}
  \end{itemize}
\end{frame}

\begin{frame}[label=parser_finiteness]

\subsection{$LR(0)$ Adequacy}

\begin{frame}
  \frametitle{$LR(0)$ Adequate and Inadequate States}
  \begin{definition}[Shift-reduce conflict]
    Let $q$ be a state of \parser. $q$ has a \concept{shift-reduce conflict} if 
    it allows for \parser\ to perform both a shift and a reduce transition.
  \end{definition}
  \begin{definition}[Reduce-reduce conflict]
    Let $q$ be a state of \parser. $q$ has a \concept{reduce-reduce conflict} if 
    reducing transitions according to two distinct productions are possible.
  \end{definition}
  \begin{definition}[$LR(0)$-adequacy]
    A state of \parser\ is \concept{$LR(0)$-inadequate} if it has a shift-reduce
    or a reduce-reduce conflict. Otherwise, it is \concept{$LR(0)$-adequate}.
  \end{definition}
\end{frame}

\begin{frame}
  \frametitle{$LR(0)$-Adequate Cases}
  \begin{lemma}[Cases of $LR(0)$-adequate states]\label{lr_cases}
    If a state $q$ of \parser\ is $LR(0)$-adequate, one of the following holds:
    \begin{enumerate}
      \item $completes\ q = \emptyset$
      \item $q = \{\cfitem{A}{\alpha}{}\}$ for some $A$ and $\alpha$.
      \item $completes\ q = \{\cfitem{A}{}{}\}$, and every incomplete item
      $i \in q$ is of the form
      $i = \cfitem{X}{\alpha}{Y \beta}$ such that every rightmost derivation
      of a word $\derivers{P'}{Y}{\gamma} \derivera w$
      implies $\gamma = A w$.
    \end{enumerate}
  \end{lemma}
  \begin{itemize}
    \item Stated by Wilhelm et al.\ without proof
  \end{itemize}
\end{frame}

\subsection{$LR(k)$ Grammars}

\begin{frame}
  \frametitle{$LR(k)$ Grammars}
  \begin{definition}[$LR(k)$ Grammar]
    $G'$ is an \concept{$LR(k)$ grammar} if 
    \begin{gather*}
      \derivers{P'}{S'}{\alpha X w} \derivera \alpha \beta w \text{ and}\\
      \derivers{P'}{S'}{\gamma Y x} \derivera \alpha \beta y \text{ and}\\
      w|_k = y|_k
    \end{gather*}
    implies $\alpha = \gamma$, $X = Y$, and $x = y$.
  \end{definition}
  \begin{alertblock}{Remark}
    This definition is restricted to the extension of a grammar.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{$LR(k)$ Grammars}
  \note{Explain: this allows assumption WLOG about length}
  \begin{lemma}\label{lrk_wlog}
    In order to prove that $G'$ is $LR(k)$ it suffices to prove:\\
    \begin{align*}
      &\derivers{P'}{S'}{\alpha X w}
        \derivera \alpha \beta w \\
      {} &\land \derivers{P'}{S'}{\gamma Y x}
        \derivera \alpha \beta y\\
      {} &\land \gamma \delta x = \alpha \beta y\\
      {} &\land |\alpha \beta| \leq |\gamma \delta|\\
      {} &\land w|_k = y|_k
    \end{align*}
    implies $\alpha = \gamma$, $X = Y$, and $x = y$.
  \end{lemma}
\end{frame}

\begin{frame}
  \frametitle{Theorem 3.4.2: Equivalence of the $LR(0)$ Condition and
    $LR(0)$ Adequacy}
  \begin{theorem}\label{lr0_iff_no_inad}
    $G'$ is an $LR(0)$ grammar if and only if \lrdfa\ has no $LR(0)$-inadequate states.
  \end{theorem}
  \begin{itemize}
    \item ``$\implies$'' correct
    \item ``$\impliedby$'': multiple errors
    \begin{itemize}
      \item Rewritten using Lemmas~\ref{lrk_wlog} and~\ref{lr_cases}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Preservation of the $LR(k)$ Condition in Extended Grammars}
  \note{
    \begin{itemize}
      \item Grammars with left recursions standard; LR(0) problematic and needs care
      \item Parser nondeterministic for unextended LR(0) grammars with such left recursions
      \item Cycle = ambiguous grammar
      \begin{itemize}
        \item Not used in practice
        \item LR(k+1) not problematic
      \end{itemize}
    \end{itemize}
  }
  \begin{itemize}
    \item Original $LR(k)$ definition for extended grammars only
    \item For CFG $G$ with extension $G'$: $G\ is\ LR(k) \overset{?}{=} G'\ is\ LR(k)$
    does not hold in general
  \end{itemize}
  \begin{theorem}[Preservation of the $LR(0)$ condition]
    $G'$ is an $LR(0)$ grammar if and only if $G$ is an $LR(0)$ grammar with no
    left recursions of the form \derivern{P}{S}{n+1}{Nt\ S \cons \alpha}.
  \end{theorem}
  \begin{theorem}[Preservation of the $LR(k)$ condition for $k > 0$]
    $G'$ is an $LR(k)$ grammar for $k > 0$ if and only if $G$ is an $LR(k)$
    grammar with no cycles of the form \derivern{P}{S}{n+1}{S}.
  \end{theorem}
\end{frame}

\subsection{Correctness}

\begin{frame}
  \frametitle{Stack Words}
  \begin{itemize}
    \item Based on an intuitive explanation by Wilhelm et al.\ of the parser's
    stack
    \item Symbols related to stack states tracked as a string
    \item \sword{\alpha}{c_0}{c_1}: $c_0$ reaches $c_1$ with
    \concept{stack word} $\alpha$
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Stack Words: Definition}
  \begin{gather*}
    \inferrule[refl]{}{\sword{\varepsilon}{c_0}{c_0}}\\[10 pt]
    \inferrule[step]{\sword{\alpha}{c_0}{(\varrho q \sigma, u)} \\
    (\varrho q \sigma, u) \vdash_P (\varDelta_G(q, X) q \sigma, v)}{
      \sword{X ({}_{|\varrho|}|\alpha)}{c_0}{(\varDelta_G(q, X) q \sigma, v)}
    }
  \end{gather*}
\end{frame}

\begin{frame}
  \frametitle{Proving Soundness with Stack Words}
  \begin{lemma}[Stack word invariant]
    If \sword{\alpha}{([init\ \parser], u @ v)}{(q \cons qs, v)} and $q \neq
    \emptyset$, then \derivers{P'}{rev\ \alpha}{map\ Tm\ u}.
  \end{lemma}
  \begin{theorem}[Soundness of the $LR(0)$ parser]
    Every word accepted by \parser\ is a word in $L(G)$.
  \end{theorem}
\end{frame}

\begin{frame}
  \frametitle{The Shift-Reduce Pushdown Automaton}
\begin{definition}
  The \concept{shift-reduce pushdown automaton} (SRPDA) for $G$ is the
  GPDA \srpda\ with alphabet $V_T$ and
  \begin{align*}
    states &= \sym \cup \{q_0, q_f\} \text{ for } (\sym) \cap \{q_0, q_f\} = \emptyset \\
    init &= q_0\\
    final &= \{q_f\}\\
    nxt &= \ldots\\
    eps &= \ldots
  \end{align*}
\end{definition}
\begin{itemize}
  \item We denote SRPDA steps by $\vdash_M$, $\vdash_M^*$, and $\vdash_M^n$
\end{itemize}
\end{frame}

\begin{frame}
  \frametitle{\srpda\ Transitions}
  \begin{itemize}
    \item \concept{Reading} transitions
    \[ (\alpha, aw) \vdash_M (a \alpha, w) \text{ for } \alpha \neq \varepsilon \]
    \item \concept{Reducing} transitions
      \[ (rev(\alpha)\beta, w) \vdash_M (A \beta, w) \text{ for } \beta \neq \varepsilon 
      \text{ if } (A, \alpha) \in P \]
    \item \concept{Finishing} transitions
      \[ (S q_0 \alpha, w) \vdash_M (q_f \alpha, w) \]
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Proving Completeness with the SRPDA}
  \begin{lemma}[SRPDA completeness]
    If a word is in $L(G)$, it is accepted by \srpda.
  \end{lemma}
  \begin{lemma}
    If $(q_0, u) \vdash_M^* (rev(\alpha) q_0, v)$ and 
    $valids(rev(\alpha)) \neq \emptyset$, there exists a stack $\varrho$ such that
    \sword{\alpha}{(q_{0,G}, u)}{(valids(rev(\alpha)) \varrho, v)}.
  \end{lemma}
  \begin{theorem}[Completeness of the $LR(0)$ Parser]
    Every word in $L(G)$ is accepted by \parser.
  \end{theorem}
\end{frame}

\begin{frame}
  \frametitle{Correctness}
  \begin{theorem}
    The language accepted by the $LR(0)$ parser is exactly the language of G.
  \end{theorem}
\end{frame}
